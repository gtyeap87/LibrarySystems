BEGIN TRANSACTION;
CREATE TABLE [Libraries] (
    [Id] uniqueidentifier NOT NULL,
    [Location] nvarchar(max) NOT NULL,
    CONSTRAINT [PK_Libraries] PRIMARY KEY ([Id])
);

CREATE TABLE [Books] (
    [Id] uniqueidentifier NOT NULL,
    [LibraryId] uniqueidentifier NOT NULL,
    [Genre] int NOT NULL DEFAULT 0,
    [Name] nvarchar(max) NOT NULL,
    [RowVersion] rowversion NULL,
    CONSTRAINT [PK_Books] PRIMARY KEY ([Id]),
    CONSTRAINT [FK_Books_Libraries_LibraryId] FOREIGN KEY ([LibraryId]) REFERENCES [Libraries] ([Id]) ON DELETE CASCADE
);

CREATE TABLE [Members] (
    [Id] uniqueidentifier NOT NULL,
    [LibraryId] uniqueidentifier NOT NULL,
    [Name] nvarchar(max) NOT NULL,
    [JoinedDate] date NOT NULL,
    [RowVersion] rowversion NULL,
    CONSTRAINT [PK_Members] PRIMARY KEY ([Id]),
    CONSTRAINT [FK_Members_Libraries_LibraryId] FOREIGN KEY ([LibraryId]) REFERENCES [Libraries] ([Id]) ON DELETE CASCADE
);

CREATE TABLE [BookStocks] (
    [Id] uniqueidentifier NOT NULL,
    [BookId] uniqueidentifier NOT NULL,
    [Quantity] int NOT NULL,
    [RowVersion] rowversion NULL,
    CONSTRAINT [PK_BookStocks] PRIMARY KEY ([Id]),
    CONSTRAINT [FK_BookStocks_Books_BookId] FOREIGN KEY ([BookId]) REFERENCES [Books] ([Id]) ON DELETE CASCADE
);

CREATE TABLE [LoanBooks] (
    [Id] uniqueidentifier NOT NULL,
    [BookId] uniqueidentifier NOT NULL,
    [MemberId] uniqueidentifier NOT NULL,
    [LoanedDate] date NOT NULL,
    [ReturnedDate] date NULL,
    [RowVersion] rowversion NULL,
    CONSTRAINT [PK_LoanBooks] PRIMARY KEY ([Id]),
    CONSTRAINT [FK_LoanBooks_Books_BookId] FOREIGN KEY ([BookId]) REFERENCES [Books] ([Id]) ON DELETE NO ACTION,
    CONSTRAINT [FK_LoanBooks_Members_MemberId] FOREIGN KEY ([MemberId]) REFERENCES [Members] ([Id]) ON DELETE NO ACTION
);

CREATE INDEX [IX_LoanBooks_BookId] ON [LoanBooks] ([BookId]);

CREATE INDEX [IX_LoanBooks_MemberId] ON [LoanBooks] ([MemberId]);

CREATE INDEX [IX_Members_LibraryId] ON [Members] ([LibraryId]);
